import pandas as pd

# Create the dataframe
data = {
    'home team': ['footscray', 'fitzroy', 'essendon', 'carlton', 'hawthorn', 'collingwood'],
    'home team score': ['14.7 (91)', '16.14 (110)', '13.12 (90)', '20.8 (128)', '19.14 (128)', '10.13 (73)'],
    'away team': ['st kilda', 'north melbourne', 'richmond', 'south melbourne', 'geelong', 'melbourne'],
    'away team score': ['9.11 (65)', '9.12 (66)', '17.9 (111)', '8.15 (63)', '15.8 (98)', '8.10 (58)'],
    'venue': ['western oval', 'junction oval', 'windy hill', 'princes park', 'glenferrie oval', 'vfl park'],
    'crowd': [18655, 7007, 22251, 14465, 12425, 30883],
    'date': ['15 july 1972', '15 july 1972', '15 july 1972', '15 july 1972', '15 july 1972', '15 july 1972']
}
df = pd.DataFrame(data)

# Extract numeric score from 'away team score' column
def extract_score(score_str):
    # Format is like "9.11 (65)" or "8.10 (58)", we need the number inside parentheses
    import re
    match = re.search(r'\((\d+)\)', score_str)
    if match:
        return int(match.group(1))
    else:
        return None

df['away_score_numeric'] = df['away team score'].apply(extract_score)

# Filter rows where 'away team' is 'melbourne'
melbourne_away = df[df['away team'] == 'melbourne']

if melbourne_away.empty:
    answer = False
else:
    # Get the numeric score for Melbourne as away team
    melbourne_score = melbourne_away['away_score_numeric'].iloc[0]
    
    # Get all away scores and rank them
    all_away_scores = df['away_score_numeric'].tolist()
    # Sort in ascending order (lowest first)
    sorted_scores = sorted(all_away_scores)
    
    # Check if melbourne_score is the second lowest
    # Need to handle ties: if multiple teams have the same lowest score, 
    # they would all be considered first lowest, then next distinct score is second lowest
    unique_sorted = sorted(set(sorted_scores))
    if len(unique_sorted) >= 2:
        second_lowest_score = unique_sorted[1]
        answer = (melbourne_score == second_lowest_score)
    else:
        # If there's only one distinct score, Melbourne cannot be second lowest
        answer = False

answer
