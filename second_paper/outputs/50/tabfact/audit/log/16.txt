import pandas as pd

# Create the dataframe from the provided table
data = {
    'player': ['adrian dantley', 'brad davis', 'darryl dawkins', 'paul dawkins', 'greg deane', 'james donaldson', 'john drew', 'john duren'],
    'nationality': ['united states', 'united states', 'united states', 'united states', 'united states', 'united states', 'united states', 'united states'],
    'position': ['guard - forward', 'guard', 'center', 'guard', 'guard', 'center', 'guard - forward', 'guard'],
    'years for jazz': ['1979 - 86', '1979 - 80', '1987 - 88', '1979 - 80', '1979 - 80', '1993 , 1994 - 95', '1982 - 85', '1980 - 82'],
    'school / club team': ['notre dame', 'maryland', 'maynard evans hs', 'northern illinois', 'utah', 'washington state', 'gardner - webb', 'georgetown']
}
df = pd.DataFrame(data)

# Function to parse years string into a list of years
def parse_years(years_str):
    years = set()
    parts = years_str.replace(' ', '').split(',')
    for part in parts:
        if '-' in part:
            start, end = part.split('-')
            years.update(range(int(start), int(end) + 1))
        else:
            years.add(int(part))
    return sorted(years)

# Get years for adrian dantley and brad davis
dantley_years = parse_years(df.loc[df['player'] == 'adrian dantley', 'years for jazz'].iloc[0])
davis_years = parse_years(df.loc[df['player'] == 'brad davis', 'years for jazz'].iloc[0])

# Check if dantley was on the team the entire time davis was also on the team
davis_years_subset = set(davis_years).issubset(set(dantley_years))

# Check if dantley surpasses davis by four additional years
dantley_additional_years = len(set(dantley_years) - set(davis_years))
surpass_by_four = dantley_additional_years == 4

# Final answer: both conditions must be true
answer = davis_years_subset and surpass_by_four
