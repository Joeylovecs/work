# Convert the 'Date' column to a datetime-like format for sorting, using the first date in the range
def extract_first_date(date_str):
    # Split by commas and take the first part, then split by '–' or '-' and take the first part
    first_part = date_str.split(',')[0]
    # Remove any leading/trailing spaces
    first_part = first_part.strip()
    # Split by range indicators
    if '–' in first_part:
        first_date = first_part.split('–')[0]
    elif '-' in first_part:
        first_date = first_part.split('-')[0]
    else:
        first_date = first_part
    # Append a year (2011) to make it a full date
    return pd.to_datetime(first_date + ' 2011', format='%b %d %Y', errors='coerce')

df['FirstDate'] = df['Date'].apply(extract_first_date)
