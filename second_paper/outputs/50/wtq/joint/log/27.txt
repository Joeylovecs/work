import pandas as pd

# Create the DataFrame from the provided table data
data = {
    'Ship': ['Argus', 'James Carruthers', 'Hydrus', 'Leafield', 'John A. McGean', 'Plymouth', 'Charles S. Price', 'Regina', 'Issac M. Scott', 'Henry B. Smith', 'Wexford', 'Lightship No. 82'],
    'Type of Vessel': ['Steamer', 'Steamer', 'Steamer', 'Steamer', 'Steamer', 'Barge', 'Steamer', 'Steamer', 'Steamer', 'Steamer', 'Steamer', 'Lightship'],
    'Lake': ['Lake Huron', 'Lake Huron', 'Lake Huron', 'Lake Superior', 'Lake Huron', 'Lake Michigan', 'Lake Huron', 'Lake Huron', 'Lake Huron', 'Lake Superior', 'Lake Huron', 'Lake Erie'],
    'Location': ['25 miles off Kincardine, Ontario', 'near Kincardine', 'near Lexington, Michigan', '', 'near Goderich, Ontario', '', 'near Port Huron, Michigan', 'near Harbor Beach, Michigan', 'near Port Elgin, Ontario', '', 'north of Grand Bend, Ontario', 'Point Albino (near Buffalo)'],
    'Lives lost': ['25 lost', '18 lost', '28 lost', 'all hands', '28 lost', '7 lost', '28 lost', '', '28 lost', 'all hands', 'all hands', '6 lost']
}
df = pd.DataFrame(data)

# Filter for barges
barge_df = df[df['Type of Vessel'] == 'Barge']

# Function to extract numeric lives lost from the 'Lives lost' column
def extract_lives(lost_str):
    if pd.isna(lost_str) or lost_str == '':
        return 0
    # Check for 'all hands' - treat as unknown, but since we need >5, we'll treat as 0 for safety (since we can't assume >5)
    if 'all hands' in lost_str.lower():
        return 0  # Not numeric, so cannot guarantee >5
    # Extract the first number from the string
    import re
    numbers = re.findall(r'\d+', lost_str)
    if numbers:
        return int(numbers[0])
    else:
        return 0

# Apply the function to create a numeric column
barge_df = barge_df.copy()
barge_df['Numeric_Lives'] = barge_df['Lives lost'].apply(extract_lives)

# Filter for ships with more than 5 lives lost
filtered = barge_df[barge_df['Numeric_Lives'] > 5]

# Get the ship name(s)
answer = filtered['Ship'].tolist()
