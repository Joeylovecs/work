# Create a function to extract end year from the 'Held record' string
def get_end_year(record):
    if pd.isna(record) or record == '':
        return None
    # Split by '–' and take the second part
    parts = record.split('–')
    if len(parts) < 2:
        return None
    end_part = parts[1]
    # If the end part is 'present', return None as it's the current record
    if 'present' in end_part.lower():
        return None
    # Otherwise, extract the year (might be with additional text, but in this table it's just the year)
    # We'll take the first four digits if they exist
    import re
    year_match = re.search(r'\d{4}', end_part)
    if year_match:
        return year_match.group()
    else:
        return None

# Apply the function to get end years
