# Convert height to a numeric value in inches for comparison
def height_to_inches(height_str):
    # Extract the part before 'cm' which contains the height in cm and the feet/inches in parentheses
    # The format is like "170 cm (5 ft 7 in)"
    # We'll parse the feet and inches from the parentheses
    import re
    # Find the pattern for feet and inches
    match = re.search(r'\((\d+)\s*ft\s*(\d+(?:\.\d+)?)\s*in\)', height_str)
    if match:
        feet = float(match.group(1))
        inches = float(match.group(2))
        return feet * 12 + inches
    else:
        return None

# Apply the conversion to the Height column
df['height_inches'] = df['Height'].apply(height_to_inches)

# Define the threshold: 5 ft 11 inches = 5*12 + 11 = 71 inches

# Filter rows where height_inches > 71
taller_models = df[df['height_inches'] > threshold_inches]

# Count the number of such models
answer = len(taller_models)
