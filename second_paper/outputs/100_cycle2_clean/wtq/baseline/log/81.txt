# Extract the 'Structural type' column and filter for rows where it contains "Tower"

# Assuming the 'Held record' format is like "1889–1930" or "2007–present"
def extract_start_year(record):
    if isinstance(record, str):
        # Split by '–' and take the first part, remove any non-digit characters
        start_part = record.split('–')[0].strip()
        # Extract digits, assuming the start part is a year (possibly with BC)
        # For BC years like "2570 BC", we'll keep as negative? But for sorting we need numeric.
        # Since the question is about "first to reach new heights", we need chronological order.
        # We'll convert BC years to negative numbers for sorting.
        if 'BC' in start_part:
            year_num = -int(start_part.replace('BC', '').strip())
        else:
            # Remove any non-digit characters (like spaces, commas) and convert to int
            year_num = int(''.join(filter(str.isdigit, start_part)))
        return year_num
    else:
        return None

tower_rows = tower_rows.copy()
