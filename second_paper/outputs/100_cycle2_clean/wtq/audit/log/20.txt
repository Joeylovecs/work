===================Question===================

===================Text===================
~~~python~~~
# Extract the record column and find consecutive years with the same record
records = df['Record'].astype(str)  # Ensure string type for comparison
years = df['Season'].astype(int)

# Initialize variables to track consecutive same records
consecutive_count = 1
start_year = years.iloc[0]
current_record = records.iloc[0]
found_sequences = []

# Iterate through the records to find sequences of three consecutive years with the same record
for i in range(1, len(records)):
    if records.iloc[i] == current_record:
        consecutive_count += 1
        if consecutive_count == 3:
            # Found a sequence of three consecutive years
            found_sequences.append((start_year, years.iloc[i]))
    else:
        # Reset the sequence
        consecutive_count = 1
        start_year = years.iloc[i]
        current_record = records.iloc[i]

# Extract the years from the found sequences
answer = []
for start, end in found_sequences:
    # Generate the three consecutive years
    answer.extend(list(range(start, end + 1)))

# Remove duplicates and sort
answer = sorted(set(answer))

# If no sequence found, answer remains an empty list
