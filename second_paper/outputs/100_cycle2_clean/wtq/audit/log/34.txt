# First, extract the date for Fantasia Festival to use as a reference
fantasia_date = df.loc[df['Festival'] == 'Fantasia Festival', 'Date'].iloc[0]

# Convert date strings to a comparable format. Since dates are in a range format like "Jul 18, Jul 25",
# we can take the earliest date in the range for comparison.
# We'll parse the first part of the date string (before any comma or dash) to get a month and day.
def parse_first_date(date_str):
    # Split by common separators: comma, dash, or space
    # Take the first token that looks like a month abbreviation followed by a day
    parts = date_str.replace(',', ' ').replace('–', ' ').split()
    # Find the first part that is a month abbreviation (three letters)
    months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec']
    for i, part in enumerate(parts):
        if part in months:
            # The next part should be the day (might have a comma or other punctuation)
            day = parts[i+1].strip(',')
            # Return a tuple (month, day) for easy comparison
            return (months.index(part), int(day))
    # If no valid month found, return a large value for sorting purposes
    return (12, 31)

# Parse the Fantasia Festival date
fantasia_parsed = parse_first_date(fantasia_date)

# Now, filter festivals that occurred before Fantasia Festival
# We'll compare based on the parsed date
before_festivals = []
for idx, row in df.iterrows():
    festival_date = row['Date']
    festival_parsed = parse_first_date(festival_date)
    if festival_parsed < fantasia_parsed:
        before_festivals.append(row['Festival'])

# The question asks for the festival before Fantasia Festival, implying the immediate predecessor.
# Since the table is likely sorted by date, we can take the last festival before Fantasia.
# But to be safe, we'll find the festival with the latest date that is still before Fantasia.
# We'll create a temporary DataFrame for comparison
temp_df = df.copy()
temp_df['parsed_date'] = temp_df['Date'].apply(parse_first_date)
# Filter rows before Fantasia
before_df = temp_df[temp_df['parsed_date'] < fantasia_parsed]
# Sort by parsed_date to get the closest one before Fantasia
before_df = before_df.sort_values('parsed_date', ascending=False)
# The festival immediately before Fantasia is the first row in this sorted DataFrame
answer = before_df.iloc[0]['Festival'] if not before_df.empty else None
