    prev_wins, prev_losses = 23, 16  # from May 24 row: record is 23 - 16

        outcomes.append('loss')
    elif new_wins > current_wins:
        outcomes.append('win')
    else:
        outcomes.append('tie')  # not expected
    current_wins, current_losses = new_wins, new_losses

        consecutive_losses += 1
        max_consecutive = max(max_consecutive, consecutive_losses)
    else:
        consecutive_losses = 0
