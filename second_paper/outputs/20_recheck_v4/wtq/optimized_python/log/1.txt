# Define a function to convert time string to total seconds for comparison
def time_to_seconds(time_str):
    if time_str == 's.t.':
        # "s.t." means same time as leader, so return leader's seconds
        return time_to_seconds(leader_time_str)
    # Handle format like "5h 29' 10\""
    # Remove any extra spaces and split by h, ', "
    time_str = time_str.strip()
    hours = 0
    minutes = 0
    seconds = 0
    if 'h' in time_str:
        hours_part, rest = time_str.split('h')
        hours = int(hours_part.strip())
        time_str = rest.strip()
    if "'" in time_str:
        minutes_part, rest = time_str.split("'")
        minutes = int(minutes_part.strip())
        time_str = rest.strip()
    if '"' in time_str:
        seconds = int(time_str.replace('"', '').strip())
    return hours * 3600 + minutes * 60 + seconds

# Apply conversion to create a numeric time column
df['time_seconds'] = df['Time'].apply(time_to_seconds)
# Find the cyclist with the minimum time (first to finish)
