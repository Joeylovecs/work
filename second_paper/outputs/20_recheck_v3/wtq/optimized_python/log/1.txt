# Find the cyclist with the minimum time (first to finish)
# Convert Time column to a sortable format (seconds) for ranking
def time_to_seconds(t):
    if isinstance(t, str):
        if t == "s.t.":
            return 0  # same time as leader
        elif "+" in t:
            # format like "+ 2""
            sec = int(t.split()[1].replace('"', ''))
            return sec
        else:
            # format like "5h 29' 10""
            parts = t.replace('h', ' ').replace("'", ' ').replace('"', '').split()
            h = int(parts[0])
            m = int(parts[1])
            s = int(parts[2])
            return h*3600 + m*60 + s
    return t

df['Time_seconds'] = df['Time'].apply(time_to_seconds)
# Find the cyclist with minimum time (first to finish)
first_cyclist = df.loc[df['Time_seconds'].idxmin(), 'Cyclist']
answer = first_cyclist
