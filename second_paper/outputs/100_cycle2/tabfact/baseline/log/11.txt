# Extract away team scores and convert to numeric values
# The score format is like "9.11 (65)" where the number in parentheses is the total points
# We need to extract the numeric total from parentheses
df['away_score_numeric'] = df['away team score'].str.extract(r'\((\d+)\)').astype(int)

# Get unique away teams
unique_away_teams = df['away team'].unique()

# Count how many unique away teams have at least one game with score > 10
teams_with_score_gt_10 = set()
for team in unique_away_teams:
    team_scores = df[df['away team'] == team]['away_score_numeric']
    if (team_scores > 10).any():
        teams_with_score_gt_10.add(team)

# Calculate the fraction
fraction = len(teams_with_score_gt_10) / len(unique_away_teams)

# Check if fraction is exactly 1/3 (within floating point tolerance)
answer = abs(fraction - 1/3) < 1e-10
