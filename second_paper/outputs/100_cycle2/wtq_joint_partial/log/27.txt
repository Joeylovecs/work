# Filter rows where 'Type of Vessel' is 'Barge' and 'Lives lost' contains a number greater than 5
# First, extract numeric part from 'Lives lost' column, handling 'all hands' and empty strings
def extract_lives(lost_str):
    if isinstance(lost_str, str):
        # Remove any leading/trailing whitespace
        lost_str = lost_str.strip()
        if lost_str == '' or lost_str == 'all hands':
            return None
        # Extract the first number found
        import re
        match = re.search(r'(\d+)', lost_str)
        if match:
            return int(match.group(1))
    return None

# Apply the function to create a temporary numeric column
df['lives_numeric'] = df['Lives lost'].apply(extract_lives)

# Filter for barges with lives lost > 5
barge_over_5 = df[(df['Type of Vessel'] == 'Barge') & (df['lives_numeric'] > 5)]

# Get the ship name(s)
answer = barge_over_5['Ship'].tolist()

# Clean up: remove the temporary column
