def parse_conference_record(record_str):
    if pd.isna(record_str):
        return None
    # Split by '-' and convert parts to integers
    parts = record_str.split('–')
    if len(parts) != 2:
        return None
    try:
        wins = int(parts[0])
        losses = int(parts[1])
        return wins, losses
    except:
        return None

# Apply parsing to get losses
