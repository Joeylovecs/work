# Extract the height in feet and inches from the 'Height' column
# The height format is like "170 cm (5 ft 7 in)" or "181 cm (5 ft 11.25 in)"
# We need to parse the feet and inches part.

def parse_height_in_inches(height_str):
    # Find the part inside parentheses
    import re
    match = re.search(r'\(([^)]+)\)', height_str)
    if match:
        ft_in = match.group(1)
        # Split by ft and in
        parts = ft_in.split('ft')
        if len(parts) == 2:
            ft = float(parts[0].strip())
            # Remove 'in' and any extra spaces
            inches_part = parts[1].replace('in', '').strip()
            # Some inches have fractions like '11.25'
            inches = float(inches_part)
            total_inches = ft * 12 + inches
            return total_inches
    return None
