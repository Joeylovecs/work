    if pd.isna(record_str):

    # Split by '–' and take the second part if it exists
    parts = str(record_str).split('–')
    if len(parts) == 2:
        end_part = parts[1]
        # If end part is 'present', return a very large number (or current year) for comparison
        if end_part.strip().lower() == 'present':
            return float('inf')
        # Otherwise, try to extract year (might be BC or AD)
        # Remove any non-digit and BC/AD indicators for simplicity
        end_part_clean = end_part.strip()
        if 'BC' in end_part_clean:
            # Convert BC year to negative integer
            year_num = -int(''.join(filter(str.isdigit, end_part_clean)))
        else:
            # AD year

        return None
