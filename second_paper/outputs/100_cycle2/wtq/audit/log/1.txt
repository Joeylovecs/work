===================Text===================
~~~python~~~
# Extract country from Cyclist column (format: "Name (COUNTRY_CODE)")
df['Country'] = df['Cyclist'].str.extract(r'\(([A-Z]{3})\)')

# Filter to top 10 ranks
top10 = df[df['Rank'] <= 10]

# Count unique cyclists per country in top 10 (though each cyclist appears once, we count distinct just in case)
country_counts = top10.groupby('Country')['Cyclist'].nunique()

# Find country with maximum count
max_country = country_counts.idxmax()

answer = max_country
