# Since the years are strings, we need to find the maximum year among Umbro entries
# The years are in format like '1982–1985', we can extract the start year for ordering
def extract_start_year(year_str):
    # Split by '–' and take the first part, convert to int
    return int(year_str.split('–')[0])

# Find the maximum start year among Umbro entries
max_umbro_start = max(extract_start_year(year) for year in umbro_years)

# Now find all rows with start year greater than max_umbro_start

df['Start Year'] = df['Year'].apply(extract_start_year)

# Filter rows after the last Umbro year
after_umbro = df[df['Start Year'] > max_umbro_start]

# Sort by Start Year to get the next one chronologically
after_umbro_sorted = after_umbro.sort_values('Start Year')
