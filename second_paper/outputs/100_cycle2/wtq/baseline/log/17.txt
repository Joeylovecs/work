# Function to convert a year range string like '1997–1999' into a list of individual years
def year_range_to_list(year_str):
    # Split by the en dash (or hyphen) and handle potential spaces
    parts = year_str.split('–')
    if len(parts) != 2:
        # If not a range, assume it's a single year (though not present in this data)
        return [year_str]
    start, end = parts[0], parts[1]
    # Convert to integers if possible
    try:
        start_int = int(start)
        # If end is empty (like '2008–'), treat as single year
        if end == '':
            return [start_int]
        end_int = int(end)
        # Generate list of years from start to end inclusive
        return list(range(start_int, end_int + 1))
    except ValueError:
        # If conversion fails, return the original string as a list
        return [year_str]

# Convert each year range to a list of individual years and flatten
all_years = []
for year_range in years:
    all_years.extend(year_range_to_list(year_range))

# Count unique years
unique_years = set(all_years)
total_unique_years = len(unique_years)
