~~~python~~~
# Filter rows where the 'Tournament' column is 'Win–Loss' to get the yearly win-loss records
win_loss_row = df[df['Tournament'] == 'Win–Loss']

# Extract the win-loss values for each year (columns 2004 to 2014)
# The win-loss values are in the format 'W–L', e.g., '1–1'
# We need to split on '–' and convert the first part (wins) to integer
# Then check if wins > 10 for each year
years_above_10 = []
for year in ['2004', '2005', '2006', '2007', '2008', '2009', '2010', '2011', '2012', '2013', '2014']:
    if year in win_loss_row.columns:
        win_loss_str = win_loss_row[year].iloc[0]
        # Some entries might be NaN or empty, skip them
        if isinstance(win_loss_str, str):
            wins = int(win_loss_str.split('–')[0])
            if wins > 10:
                years_above_10.append(year)

answer = years_above_10
