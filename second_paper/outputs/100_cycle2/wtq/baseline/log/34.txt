~~~
# First, extract the date for Fantasia Festival to use as a reference
fantasia_date = df.loc[df['Festival'] == 'Fantasia Festival', 'Date'].iloc[0]

# Convert date strings to a comparable format (assuming we can compare as strings for this table's date format)
# We'll parse the start date (first part before any separator) to a datetime for comparison
def get_start_date(date_str):
    # Split by common separators and take the first part
    first_part = date_str.split(',')[0].split('–')[0].strip()
    # Parse to datetime, assuming month abbreviation and day format
    # Since we don't have year, we'll assume 2011 (from the film title) for parsing
    from datetime import datetime
    try:
        return datetime.strptime(f"{first_part} 2011", "%b %d %Y")
    except ValueError:
        # Handle cases like "Feb 2–5" where first_part is "Feb 2"
        # Remove month abbreviation and keep day
        day_part = first_part.split(' ')[-1]
        month_part = first_part.split(' ')[0]
        return datetime.strptime(f"{month_part} {day_part} 2011", "%b %d %Y")

# Apply to all dates
df['StartDate'] = df['Date'].apply(get_start_date)

# Get the Fantasia Festival start date
fantasia_start = df.loc[df['Festival'] == 'Fantasia Festival', 'StartDate'].iloc[0]

# Filter festivals with start date before Fantasia's start date
before_fantasia = df[df['StartDate'] < fantasia_start]

# Sort by start date descending to get the one immediately before
before_fantasia_sorted = before_fantasia.sort_values('StartDate', ascending=False)

# Get the festival name of the one immediately before Fantasia
if not before_fantasia_sorted.empty:
    answer = before_fantasia_sorted.iloc[0]['Festival']
else:
    answer = None
