# The 'Year' column contains strings like '1931', 'Spring 1932', '1933/34', etc.
# We need to extract the numeric year part for comparison.
def extract_year(year_str):
    # Remove any leading/trailing whitespace
    year_str = str(year_str).strip()
    # If it's a simple year like '1931', return as int
    if year_str.isdigit():
        return int(year_str)
    # If it contains a slash like '1933/34', take the first part
    if '/' in year_str:
        first_part = year_str.split('/')[0]
        if first_part.isdigit():
            return int(first_part)
    # If it contains a space like 'Spring 1932', take the last part
    parts = year_str.split()
    for part in parts:
        if part.isdigit():
            return int(part)
    # If no numeric part found, return None (should not happen for this data)
    return None

# Apply the extraction to create a temporary numeric year column
