# We'll extract the first four-digit number from each Year string
def extract_year(year_str):
    # Find all sequences of digits
    import re
    digits = re.findall(r'\d+', str(year_str))
    if digits:
        # Take the first sequence that is 4 digits (or any digits if not)
        for d in digits:
            if len(d) == 4:
                return int(d)
        # If no 4-digit number, take the first sequence and interpret based on context
        # For years like '1931', 'Spring 1932', '1933/34'
        first = digits[0]
        if len(first) == 4:
            return int(first)
        elif len(first) == 2:
            # Handle two-digit years like '34' in '1933/34'
            # We'll assume it's 1900+ if less than 100, but need to get full year from string
            # Look for a 4-digit year first in the whole string
            four_digit = re.search(r'\b\d{4}\b', str(year_str))
            if four_digit:
                return int(four_digit.group())
            else:
                # If no 4-digit, assume 1900 + two-digit year
                return 1900 + int(first)
    return None

champion_rows['year_num'] = champion_rows['Year'].apply(extract_year)

    answer = earliest_year - 1936
