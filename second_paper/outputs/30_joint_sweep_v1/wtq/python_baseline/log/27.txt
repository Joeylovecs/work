# Extract the year from the 'Year' column for rows where National Cup is 'Champion'
# First, let's clean the Year column to get the start year
def extract_start_year(year_str):
    # Handle cases like 'Spring 1932', '1933/34', '1931'
    if '/' in year_str:
        return int(year_str.split('/')[0])
    else:
        # Extract the last part which should be the year
        parts = year_str.split()
        for part in parts:
            if part.isdigit():
                return int(part)
        # If no digit found, return None
        return None

# Apply the function to create a new column
df['Start_Year'] = df['Year'].apply(extract_start_year)

champion_years.sort()

# Find the first champion year after 1936

post_1936_champions = [year for year in champion_years if year > 1936]

if post_1936_champions:
    next_win_year = post_1936_champions[0]
    # Calculate years since 1936
    years_diff = next_win_year - 1936
