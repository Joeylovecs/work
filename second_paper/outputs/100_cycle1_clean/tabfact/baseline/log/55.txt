# First, we need to parse the score column to extract the scores for each game.
# The score column is in the format "visitor_score - home_score".

# Determine if Detroit is the home team or visitor in each row.
# Then compute the point gap (absolute difference in scores) for each game.
def compute_gap(row):
    if row['home'] == 'detroit':
        # Detroit is home, so their score is home_score
        detroit_score = row['home_score']
        opponent_score = row['visitor_score']
    elif row['visitor'] == 'detroit':
        # Detroit is visitor, so their score is visitor_score
        detroit_score = row['visitor_score']
        opponent_score = row['home_score']
    else:
        # Detroit not involved? Should not happen in this dataset, but handle.
        return None
    return abs(detroit_score - opponent_score)

df['point_gap'] = df.apply(compute_gap, axis=1)
