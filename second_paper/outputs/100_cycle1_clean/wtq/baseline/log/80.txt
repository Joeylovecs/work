===================Question===================
which building held the record for tallest freestanding structure before the burj khalifa?
===================Text===================
~~~python~~~
# Extract the record holder before Burj Khalifa
# First, find the row for Burj Khalifa
burj_row = df[df['Name'] == 'Burj Khalifa']
# Get the 'Held record' value for Burj Khalifa
burj_record = burj_row['Held record'].iloc[0]
# The record is in the format 'start_year–end_year' or 'start_year–present'
# We need to find the structure whose record end_year matches the start_year of Burj Khalifa
# Extract the start year of Burj Khalifa's record
burj_start = burj_record.split('–')[0]
# Now find the row where the 'Held record' end_year equals burj_start
# The 'Held record' column may have values like '2007–present', '1967–1976', etc.
# We need to check if the second part (after '–') matches burj_start
def get_end_year(record):
    parts = record.split('–')
    if len(parts) == 2:
        return parts[1]
    else:
        return None

# Apply to find the previous record holder
df['end_year'] = df['Held record'].apply(get_end_year)
previous_record_holder = df[df['end_year'] == burj_start]
# Get the name of the building
answer = previous_record_holder['Name'].iloc[0]
