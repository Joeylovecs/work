# Function to convert a year range string like "1997–1999" into a list of years
def year_range_to_list(year_str):
    # Split by the dash (note: the dash is an en dash, not a hyphen)
    parts = year_str.split('–')
    if len(parts) != 2:
        return []
    start, end = parts[0], parts[1]
    # If end is empty (e.g., "2008–"), use start as end (single year)
    if end == '':
        end = start
    # Convert to integers
    try:
        start_int = int(start)
        end_int = int(end)
    except ValueError:
        return []
    # Generate list of years from start to end inclusive
    return list(range(start_int, end_int + 1))

# Apply the function to each year range and flatten the list
all_years = []
for year_range in years:
    all_years.extend(year_range_to_list(year_range))

# Remove duplicates (in case of overlapping ranges) and count unique years
total_unique_years = len(set(all_years))
