# The height is given as "X cm (Y ft Z in)", we need to parse the feet and inches part.
# We'll convert the height in feet and inches to total inches for comparison.

def feet_inches_to_inches(height_str):
    # Example: "170 cm (5 ft 7 in)" -> we want the part inside parentheses
    import re
    # Find the pattern for feet and inches
    match = re.search(r'\((\d+)\s*ft\s*(\d+(?:\.\d+)?)\s*in\)', height_str)
    if match:
        feet = float(match.group(1))
        inches = float(match.group(2))
        return feet * 12 + inches
    else:
        return None

# Apply the conversion to each row
