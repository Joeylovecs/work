# Parse the year ranges
def parse_years(years_str):
    # Handle possible multiple ranges (e.g., "1993 , 1994 - 95")
    parts = years_str.split(',')
    years = []
    for part in parts:
        part = part.strip()
        if '-' in part:
            start, end = part.split('-')
            start = start.strip()
            end = end.strip()
            # Convert to full years (assuming 1900s for two-digit years)
            start_year = int(start) if len(start) == 4 else int('19' + start)
            end_year = int(end) if len(end) == 4 else int('19' + end)
            years.extend(range(start_year, end_year + 1))
        else:
            year = int(part) if len(part) == 4 else int('19' + part)
            years.append(year)
    return set(years)

dantley_set = parse_years(dantley_years)
davis_set = parse_years(davis_years)

# Check if all years Brad Davis was on the team are within Adrian Dantley's years
answer = davis_set.issubset(dantley_set)
