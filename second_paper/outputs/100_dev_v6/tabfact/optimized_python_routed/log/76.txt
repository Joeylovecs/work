    def extract_years(year_str):
        years = set()
        # Split by commas and dashes to handle multiple ranges
        parts = year_str.replace(' ', '').split(',')
        for part in parts:
            if '-' in part:
                start, end = part.split('-')
                # Handle two-digit years (like '86') by assuming they are in the 1900s
                start = int(start)
                end = int(end)
                if end < 100:
                    # If end is two digits, assume it's 1900 + end
                    end += 1900
                if start < 100:
                    start += 1900
                years.update(range(start, end + 1))
            else:
                year = int(part)
                if year < 100:
                    year += 1900
                years.add(year)
        return years

    # Apply the function to get sets of years for each Dawkins player
    years_sets = dawkins_players['years for jazz'].apply(extract_years).tolist()

    # Check if the sets of years are disjoint (no overlap)
