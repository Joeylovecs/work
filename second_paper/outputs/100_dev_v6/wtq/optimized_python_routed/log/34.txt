# Filter rows where the 'National Cup' column contains the exact string 'Champion'
champion_cup_rows = df[df['National Cup'] == 'Champion']

# If there are multiple rows, we need to find the latest year.
# The 'Year' column contains strings, some with seasons like 'Spring 1932' or '1933/34'.
# We need to extract the ending year for comparison.
def extract_year(year_str):
    # Remove any leading/trailing whitespace
    s = str(year_str).strip()
    # Cases like 'Spring 1932' -> extract the number after space
    if ' ' in s:
        # Take the last part after the last space
        num_part = s.split()[-1]
    else:
        num_part = s
    # Cases like '1933/34' -> take the first part before '/'
    if '/' in num_part:
        base_year = num_part.split('/')[0]
    else:
        base_year = num_part
    # Convert to integer, if possible
    try:
        return int(base_year)
    except:
        # If conversion fails, return a very small number for sorting (these rows likely not relevant)
        return 0

# Apply the extraction to create a temporary numeric year column
champion_cup_rows = champion_cup_rows.copy()
champion_cup_rows['year_num'] = champion_cup_rows['Year'].apply(extract_year)

# Sort by the numeric year in descending order to get the latest
champion_cup_rows_sorted = champion_cup_rows.sort_values('year_num', ascending=False)

# The latest row's 'Year' value (original string) is the answer
if not champion_cup_rows_sorted.empty:
    answer = champion_cup_rows_sorted.iloc[0]['Year']
else:
    answer = None
