# Extract declination values and convert to decimal degrees
def declination_to_decimal(dec_str):
    # Format: degreeXX′YY″
    # Example: degree22′44″
    # Remove 'degree' and split by ′ and ″
    dec_str = dec_str.replace('degree', '').replace('′', ' ').replace('″', '')
    parts = dec_str.split()
    degrees = float(parts[0])
    minutes = float(parts[1]) if len(parts) > 1 else 0
    seconds = float(parts[2]) if len(parts) > 2 else 0
    return degrees + minutes/60 + seconds/3600

# Apply conversion
