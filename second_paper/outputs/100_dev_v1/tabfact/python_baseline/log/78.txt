# Determine if Adrian Dantley was on the team the entire time Brad Davis was
# Extract years for both players
dantley_years = df[df['player'] == 'adrian dantley']['years for jazz'].iloc[0]
davis_years = df[df['player'] == 'brad davis']['years for jazz'].iloc[0]

# Parse year ranges
def parse_years(years_str):
    # Handle cases like "1993 , 1994 - 95" by taking the full range
    parts = years_str.replace(' ', '').split(',')
    all_years = []
    for part in parts:
        if '-' in part:
            start, end = part.split('-')
            # Handle two-digit years
            start_year = int(start)
            end_year = int(end) if len(end) == 4 else int(start[:2] + end)
            all_years.extend(range(start_year, end_year + 1))
        else:
            all_years.append(int(part))
    return set(all_years)

dantley_set = parse_years(dantley_years)
davis_set = parse_years(davis_years)

# Check if all of Brad Davis's years are in Adrian Dantley's years
answer = davis_set.issubset(dantley_set)
answer
