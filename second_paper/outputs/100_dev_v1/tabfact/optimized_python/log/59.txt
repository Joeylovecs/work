# Extract declination values and convert to numeric degrees for comparison
# The declination column contains strings like " degree22′44″"
# We need to parse degrees, minutes, seconds and convert to decimal degrees

def parse_declination(dec_str):
    # Remove leading/trailing whitespace
    dec_str = str(dec_str).strip()
    # Extract numeric parts
    # Format: " degree22′44″" or similar
    # We'll split by non-digit characters
    import re
    # Find all numbers in the string
    numbers = re.findall(r'\d+', dec_str)
    if len(numbers) >= 3:
        degrees = int(numbers[0])
        minutes = int(numbers[1])
        seconds = int(numbers[2])
    elif len(numbers) == 2:
        degrees = int(numbers[0])
        minutes = int(numbers[1])
        seconds = 0
    elif len(numbers) == 1:
        degrees = int(numbers[0])
        minutes = 0
        seconds = 0
    else:
        return None
    # Calculate decimal degrees
    decimal_degrees = degrees + minutes/60 + seconds/3600
    return decimal_degrees

# Apply parsing to create a new column with numeric declination
df['declination_numeric'] = df['declination ( j2000 )'].apply(parse_declination)

# Filter for globular clusters
globular_df = df[df['object type'] == 'globular cluster'].copy()

# Check if there are any globular clusters
if len(globular_df) == 0:
    answer = False
else:
    # Get declination values for all objects
    all_declinations = df['declination_numeric'].dropna()
    
    # Sort declinations in descending order (highest first)
    sorted_declinations = all_declinations.sort_values(ascending=False)
    
    # Get the second highest declination value
    if len(sorted_declinations) >= 2:
        second_highest_declination = sorted_declinations.iloc[1]
        
        # Check if any globular cluster has this declination
        globular_declinations = globular_df['declination_numeric'].values
        
        # Compare with tolerance for floating point precision
        import numpy as np
        matches = np.isclose(globular_declinations, second_highest_declination, rtol=1e-10, atol=1e-10)
        
        answer = bool(np.any(matches))
    else:
        answer = False
