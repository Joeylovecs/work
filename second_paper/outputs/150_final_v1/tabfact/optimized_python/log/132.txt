# The 'relative value' column contains strings like '1 / 1600', '1', '100', etc.
# We need to evaluate these to numeric values
def parse_relative_value(val):
    if isinstance(val, str):
        if '/' in val:
            num, denom = val.split('/')
            return float(num.strip()) / float(denom.strip())
        else:
            return float(val)
    else:
        return float(val)

numeric_values = relative_values.apply(parse_relative_value)

# Check if all numeric values are at least 1
answer = all(value >= 1 for value in numeric_values)
