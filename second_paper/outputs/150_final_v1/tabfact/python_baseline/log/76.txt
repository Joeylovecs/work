def parse_years(years_str):
    # Handle cases like "1993 , 1994 - 95"
    years_str = str(years_str).replace(' ', '')
    # Find all 4-digit years
    import re
    years = re.findall(r'\d{4}', years_str)
    if years:
        return [int(y) for y in years]
    return []

dawkins_df['years_list'] = dawkins_df['years for jazz'].apply(parse_years)

# Check if their time overlapped
if len(dawkins_df) == 2:
    years1 = dawkins_df.iloc[0]['years_list']
    years2 = dawkins_df.iloc[1]['years_list']
    
    # Create sets of years each player was on the team
    set1 = set()
    set2 = set()
    
    # For each year range, add all years in between

        if i + 1 < len(years1):
            start, end = years1[i], years1[i + 1]
            set1.update(range(start, end + 1))
    
    for i in range(0, len(years2), 2):
        if i + 1 < len(years2):
            start, end = years2[i], years2[i + 1]
            set2.update(range(start, end + 1))
    
    # Check for overlap
    overlap = bool(set1.intersection(set2))
