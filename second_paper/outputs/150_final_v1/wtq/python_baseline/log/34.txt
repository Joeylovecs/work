===================Question===================

===================Text===================
```python
# First, let's filter the DataFrame to rows where National Cup is 'Champion'
champion_rows = df[df['National Cup'] == 'Champion']

# If there are multiple champion years, we need to find the latest one
# We'll extract the year from the 'Year' column and find the maximum
# However, the 'Year' column contains various formats like '1931', 'Spring 1932', '1933/34', etc.
# We need to handle these different formats to extract the ending year

def extract_year(year_str):
    # Handle cases like '1931', 'Spring 1932', '1933/34'
    if isinstance(year_str, str):
        # Remove any leading/trailing whitespace
        year_str = year_str.strip()
        
        # Case 1: Simple year like '1931'
        if year_str.isdigit():
            return int(year_str)
        
        # Case 2: Contains slash like '1933/34'
        if '/' in year_str:
            # Take the first part before slash
            return int(year_str.split('/')[0])
        
        # Case 3: Contains space like 'Spring 1932'
        # Extract the last part which should be the year
        parts = year_str.split()
        for part in reversed(parts):
            if part.isdigit():
                return int(part)
            # Handle cases like '1932?' 
            if part[:-1].isdigit() and part[-1] == '?':
                return int(part[:-1])
    
    return None

# Apply the function to create a new column with numeric years
champion_rows = champion_rows.copy()
champion_rows['numeric_year'] = champion_rows['Year'].apply(extract_year)

# Find the maximum year
if not champion_rows.empty:
    last_champion_year = champion_rows['numeric_year'].max()
    # Get the original Year value for the last champion
    last_champion_row = champion_rows[champion_rows['numeric_year'] == last_champion_year]
    answer = last_champion_row['Year'].iloc[0]
else:
    answer = None

answer
```
