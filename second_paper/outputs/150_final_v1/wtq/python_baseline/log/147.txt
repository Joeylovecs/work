===================Question===================
what was guillermo garcía-lópez combined win and loss record from the rome, madrid and paris masters?
===================Text===================
```python
# Extract the rows for Rome Masters, Madrid Masters, and Paris Masters
selected_rows = df[df['Tournament'].isin(['Rome Masters', 'Madrid Masters', 'Paris Masters'])]

# Extract the W–L column values and parse wins and losses
def parse_wl(wl_str):
    if isinstance(wl_str, str) and '–' in wl_str:
        w, l = wl_str.split('–')
        return int(w), int(l)
    return 0, 0

# Calculate combined wins and losses
total_wins = 0
total_losses = 0
for wl in selected_rows['W–L']:
    w, l = parse_wl(wl)
    total_wins += w
    total_losses += l

# Format the answer as requested
answer = f"{total_wins}–{total_losses}"
```
