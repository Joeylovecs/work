# Find the years where Umbro was the kit manufacturer
umbro_years = df[df['Kit Manufacturer'] == 'Umbro']['Year']

# Get the maximum year (most recent) where Umbro was used
latest_umbro_year = umbro_years.max()

# Find all years after the latest Umbro year
# First, convert Year strings to sortable format for comparison
# Since years are in format like "1977–1978", we'll extract the start year
df['Start_Year'] = df['Year'].str.split('–').str[0].astype(int)

# Find the next entry after the latest Umbro year
latest_umbro_start = int(latest_umbro_year.split('–')[0])
next_entries = df[df['Start_Year'] > latest_umbro_start].sort_values('Start_Year')

# Get the kit manufacturer of the next entry
if not next_entries.empty:
    next_manufacturer = next_entries.iloc[0]['Kit Manufacturer']
