# The question asks for how many years ago the New York Americans were founded.
# The dataframe includes a 'Year' column. The earliest year in the table corresponds to the founding year.
# We need to find the earliest year, then calculate the difference from the current year.
# However, note that the 'Year' column contains values like '1931', 'Spring 1932', 'Fall 1932', '1933/34', etc.
# We need to extract the numeric year part from each entry.
# We'll create a function to extract the first four-digit number from each string.
import pandas as pd
import datetime

def extract_year(year_str):
    # If the value is already a number (int or float), convert to string
    if pd.isna(year_str):
        return None
    # Convert to string
    s = str(year_str)
    # Find the first occurrence of four consecutive digits
    import re
    match = re.search(r'\b\d{4}\b', s)
    if match:
        return int(match.group())
    else:
        # If no four-digit year found, try to find any digits (but should not happen given the data)
        # For safety, return None
        return None

# Apply the function to the 'Year' column to get numeric years
df['numeric_year'] = df['Year'].apply(extract_year)

# Find the minimum numeric year (founding year)
founding_year = df['numeric_year'].min()

# Get the current year
current_year = datetime.datetime.now().year

# Calculate how many years ago
years_ago = current_year - founding_year

# Assign the final answer
answer = years_ago
