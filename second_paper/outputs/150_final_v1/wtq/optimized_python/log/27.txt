    # Remove any leading/trailing whitespace
    s = str(year_str).strip()
    # Find the first four consecutive digits
    import re
    match = re.search(r'\d{4}', s)
    if match:
        return int(match.group())
    else:
        # If no four-digit year found, return None
        return None

champion_rows['year_int'] = champion_rows['Year'].apply(extract_year)

    answer = earliest_year - 1936
