# Function to parse a year range string like "1997–1999" into start and end years
def parse_year_range(year_str):
    # Handle cases where the year might be like "2008–" (ongoing)
    if '–' not in year_str:
        return None, None
    parts = year_str.split('–')
    start = parts[0].strip()
    end = parts[1].strip()
    # If end is empty (ongoing), we cannot compute total years, so treat as missing
    if end == '':
        return None, None
    # Convert to integers
    try:
        start_int = int(start)
        end_int = int(end)
        return start_int, end_int
    except ValueError:
        return None, None

# Calculate total years for each row
total_years = 0
for year_str in years:
    start, end = parse_year_range(year_str)
