# Filter rows where Notes contain any award nomination or win
# We need to check for any mention of an award (including wins and nominations)
# Exclude non-award statuses like 'Filming' or 'Announced'
# According to the special rule: a listed award win/title is also evidence of nomination even without the word 'Nominated'

# First, drop rows where Notes is NaN or empty string

# Define a function to check if a note string indicates an award (win or nomination)
def has_award(note):
    # Convert to string to be safe
    note_str = str(note)
    # Exclude non-award statuses
    if 'Filming' in note_str or 'Announced' in note_str:
        return False
    # Check for any award-related keywords (case-insensitive)
    # We'll look for 'Award', 'Nominated', 'SIIMA', 'Filmfare', 'Udaya', 'Karnataka State', 'Innovative'
    # This covers all awards in the table
    award_keywords = ['Award', 'Nominated', 'SIIMA', 'Filmfare', 'Udaya', 'Karnataka State', 'Innovative']
    return any(keyword.lower() in note_str.lower() for keyword in award_keywords)

# Apply the function to filter rows with awards
award_films = df_filtered[df_filtered['Notes'].apply(has_award)]

# Count unique films (by film name) that were nominated for an award
# According to the question, we count film rows whose Notes contain a named Award, once per film row.
# Since the question asks "how many films were nominated for an award", we count distinct films.
answer = award_films['Film'].nunique()
