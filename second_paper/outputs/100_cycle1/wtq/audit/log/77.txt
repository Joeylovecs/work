# Convert height to a numeric value in inches for comparison
def height_to_inches(height_str):
    # Extract the part before "cm" which contains the feet and inches in parentheses
    # Example: "170 cm (5 ft 7 in)"
    # We'll parse the inches from the parentheses
    import re
    # Find the pattern for feet and inches
    match = re.search(r'\((\d+)\s*ft\s*(\d+(?:\.\d+)?)\s*in\)', height_str)
    if match:
        feet = float(match.group(1))
        inches = float(match.group(2))
        return feet * 12 + inches
    else:
        return None

# Apply conversion to the Height column
df['height_inches'] = df['Height'].apply(height_to_inches)

# Filter models taller than 5 ft 11 inches (which is 5*12 + 11 = 71 inches)
taller_than_5_11 = df[df['height_inches'] > 71]

# Count the number of such models
answer = len(taller_than_5_11)
