# Extract the 'Structural type' column to identify towers
tower_rows = df[df['Structural type'].str.contains('Tower', case=False, na=False)]

# Sort by the 'Held record' start year to find the earliest tower
# The 'Held record' column has entries like '1889–1930', we need to extract the start year
tower_rows['record_start'] = tower_rows['Held record'].str.split('–').str[0]

# Convert to numeric where possible, handling BC years (like '2570 BC')
def parse_year(year_str):
    if 'BC' in year_str:
        return -int(year_str.replace(' BC', ''))
    else:
        return int(year_str)

tower_rows['record_start_num'] = tower_rows['record_start'].apply(parse_year)

# Find the row with the smallest (earliest) record_start_num
earliest_tower = tower_rows.loc[tower_rows['record_start_num'].idxmin()]

# Get the country of that tower
answer = earliest_tower['Country']
