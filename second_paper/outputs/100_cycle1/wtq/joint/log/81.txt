# The "Held record" column contains date ranges like "1889–1930"
# Extract the start year by splitting on '–' and taking the first part
tower_rows = tower_rows.copy()
tower_rows['Start Year'] = tower_rows['Held record'].str.split('–').str[0]

# Convert start year to numeric, handling BC years (like "2570 BC")
def parse_year(year_str):
    if 'BC' in year_str:
        return -int(year_str.replace(' BC', ''))
    else:
        return int(year_str)

tower_rows['Start Year Num'] = tower_rows['Start Year'].apply(parse_year)

# Find the row with the smallest (most ancient) start year
earliest_tower = tower_rows.loc[tower_rows['Start Year Num'].idxmin()]
