# Let's define a function to check if a string is a time (finishing time or gap time without 'lap')
def is_time(s):
    # Check if it's a string
    if not isinstance(s, str):
        return False
    # Finishing time contains ':'
    if ':' in s:
        return True
    # Gap time: starts with '+' and then a number (possibly with decimal) and no 'lap'
    if s.startswith('+'):
        # Remove the plus and strip whitespace
        rest = s[1:].strip()
        # Check if it's a number (possibly with decimal) and not containing 'lap'
        # We'll try to convert to float, if successful and 'lap' not in s, then it's a gap time
        try:
            float(rest)
            return True
        except ValueError:
            # If not a pure number, might be like '+ 1 lap' which is not a time
            return False
    return False
