# Filter players from the United States
us_players = df[df['country'].str.lower() == 'united states']

# Convert 'to par' values to numeric (handle 'e' as even = 0)
def to_par_numeric(val):
    val = str(val).strip().lower()
    if val in ('e', 'even'):
        return 0
    val = val.replace('−', '-').replace(' ', '')
    return float(val)

us_players = us_players.copy()
us_players['to_par_num'] = us_players['to par'].apply(to_par_numeric)

# Steve Stricker's score vs best (lowest) to-par among US players
stricker_score = us_players.loc[us_players['player'].str.lower() == 'steve stricker', 'to_par_num'].iloc[0]
best_us_score = us_players['to_par_num'].min()

answer = bool(stricker_score == best_us_score)
print(answer)
