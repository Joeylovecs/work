cup_wins = df[df['National Cup'].astype(str).str.strip() == 'Champion'].copy()

# Extract the ending year of each season (e.g., '1936/37' -> 1937, '1931' -> 1931)
def end_year(y):
    y = str(y).strip()
    if '/' in y:
        return int(y.split('/')[1])
    return int(y)

cup_wins['end_year'] = cup_wins['Year'].apply(end_year)

# First National Cup win after 1936
first_win = cup_wins[cup_wins['end_year'] > 1936].sort_values('end_year').iloc[0]

# How long it took from 1936 to that win
answer = int(first_win['end_year'] - 1936)
